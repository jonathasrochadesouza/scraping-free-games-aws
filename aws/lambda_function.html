#!/usr/bin/env python3
"""
Epic Games Free Games Scraper
Captura os jogos gratuitos atualmente disponíveis na Epic Games Store
Python 3.12.10
"""

import requests
from bs4 import BeautifulSoup
import json
from typing import List, Dict
import sys
from datetime import datetime


class EpicGamesScraper:
    """Scraper para jogos gratuitos da Epic Games Store"""
    
    def __init__(self):
        self.url = "https://store.epicgames.com/pt-BR/free-games"
        self.headers = {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/119.0.0.0 Safari/537.36',
            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8',
            'Accept-Language': 'pt-BR,pt;q=0.9,en-US;q=0.8,en;q=0.7',
            'Accept-Encoding': 'gzip, deflate, br',
            'Connection': 'keep-alive',
        }
    
    def fetch_page(self) -> str:
        """
        Faz a requisição HTTP e retorna o conteúdo HTML
        
        Returns:
            str: Conteúdo HTML da página
            
        Raises:
            requests.RequestException: Se houver erro na requisição
        """
        try:
            print(f"🔍 Acessando: {self.url}")
            response = requests.get(
                self.url, 
                headers=self.headers, 
                timeout=30,
                verify=True
            )
            response.raise_for_status()
            print("✅ Página carregada com sucesso!")
            return response.text
            
        except requests.Timeout:
            print("❌ Erro: Timeout ao acessar a página")
            raise
        except requests.RequestException as e:
            print(f"❌ Erro ao fazer requisição: {e}")
            raise
    
    def parse_games(self, html_content: str) -> List[Dict]:
        """
        Faz o parsing do HTML e extrai informações dos jogos
        
        Args:
            html_content: Conteúdo HTML da página
            
        Returns:
            List[Dict]: Lista de dicionários com informações dos jogos
        """
        soup = BeautifulSoup(html_content, 'lxml')
        games = []
        
        # Busca a section com class css-2u323
        section = soup.find('section', class_='css-2u323')
        
        if not section:
            print("⚠️  Section 'css-2u323' não encontrada. Tentando métodos alternativos...")
            # Método alternativo: buscar por data-testid ou outras classes comuns
            section = soup.find('section', attrs={'data-testid': 'offer-card-layout-wrapper'})
            
        if not section:
            print("⚠️  Nenhuma section de jogos encontrada. A estrutura da página pode ter mudado.")
            return games
        
        # Busca todos os cards de jogos dentro da section
        game_cards = section.find_all(['article', 'a', 'div'], attrs={
            'data-testid': lambda x: x and 'offer-card' in x if x else False
        })
        
        if not game_cards:
            # Fallback: buscar por links ou estruturas comuns
            game_cards = section.find_all('a', href=lambda x: x and '/p/' in x if x else False)
        
        print(f"🎮 Encontrados {len(game_cards)} jogos")
        
        for idx, card in enumerate(game_cards, 1):
            try:
                game_data = self._extract_game_data(card)
                if game_data:
                    games.append(game_data)
                    print(f"  {idx}. {game_data.get('title', 'Sem título')}")
            except Exception as e:
                print(f"⚠️  Erro ao processar jogo {idx}: {e}")
                continue
        
        return games
    
    def _extract_game_data(self, card) -> Dict:
        """
        Extrai dados de um card de jogo individual
        
        Args:
            card: Elemento BeautifulSoup do card
            
        Returns:
            Dict: Dicionário com informações do jogo
        """
        game_data = {}
        
        # Título do jogo
        title_elem = card.find(['h3', 'h2', 'span'], class_=lambda x: x and 'title' in x.lower() if x else False)
        if not title_elem:
            title_elem = card.find('h3') or card.find('h2')
        game_data['title'] = title_elem.get_text(strip=True) if title_elem else "Título não encontrado"
        
        # Link do jogo
        link = card.get('href') or (card.find('a').get('href') if card.find('a') else None)
        if link:
            game_data['url'] = f"https://store.epicgames.com{link}" if link.startswith('/') else link
        
        # Imagem
        img = card.find('img')
        if img:
            game_data['image'] = img.get('src') or img.get('data-src')
        
        # Preço (deve ser "Grátis")
        price_elem = card.find(['span', 'div'], class_=lambda x: x and 'price' in x.lower() if x else False)
        game_data['price'] = price_elem.get_text(strip=True) if price_elem else "Grátis"
        
        # Data de disponibilidade
        date_elem = card.find(['span', 'time', 'div'], class_=lambda x: x and ('date' in x.lower() or 'time' in x.lower()) if x else False)
        if date_elem:
            game_data['available_until'] = date_elem.get_text(strip=True)
        
        return game_data
    
    def save_to_json(self, games: List[Dict], filename: str = "epic_free_games.json"):
        """
        Salva os dados dos jogos em arquivo JSON
        
        Args:
            games: Lista de jogos
            filename: Nome do arquivo de saída
        """
        output = {
            'scraped_at': datetime.now().isoformat(),
            'total_games': len(games),
            'games': games
        }
        
        try:
            with open(filename, 'w', encoding='utf-8') as f:
                json.dump(output, f, ensure_ascii=False, indent=2)
            print(f"\n✅ Dados salvos em: {filename}")
            print(f"📊 Total de jogos: {len(games)}")
        except Exception as e:
            print(f"❌ Erro ao salvar JSON: {e}")
            raise
    
    def run(self):
        """Executa o scraper completo"""
        try:
            print("🚀 Iniciando Epic Games Scraper\n")
            
            # 1. Buscar página
            html = self.fetch_page()
            
            # 2. Fazer parsing
            games = self.parse_games(html)
            
            if not games:
                print("\n⚠️  Nenhum jogo encontrado. Possíveis motivos:")
                print("   - A estrutura do site mudou")
                print("   - Não há jogos gratuitos no momento")
                print("   - Bloqueio por rate limiting")
                return
            
            # 3. Salvar em JSON
            self.save_to_json(games)
            
            print("\n✨ Scraping concluído com sucesso!")
            
        except KeyboardInterrupt:
            print("\n\n⚠️  Operação cancelada pelo usuário")
            sys.exit(0)
        except Exception as e:
            print(f"\n❌ Erro fatal: {e}")
            sys.exit(1)


def main():
    """Função principal"""
    scraper = EpicGamesScraper()
    scraper.run()


if __name__ == "__main__":
    main()
⚠️ Observação Importante: A Epic Games usa renderização dinâmica com JavaScript. Este script básico pode não capturar todos os dados se a página depender muito de JS. Para casos mais robustos, considere usar Selenium ou Playwright para renderização completa do JavaScript.

"""
Lambda Function com Playwright para Epic Games Scraper
Otimizado para custo e performance
"""

import json
import os
from datetime import datetime
from playwright.sync_api import sync_playwright

def lambda_handler(event, context):
    """Handler principal da Lambda"""
    
    try:
        games = scrape_epic_games()
        
        return {
            'statusCode': 200,
            'headers': {
                'Content-Type': 'application/json',
                'Access-Control-Allow-Origin': '*'
            },
            'body': json.dumps({
                'scraped_at': datetime.now().isoformat(),
                'total_games': len(games),
                'games': games
            }, ensure_ascii=False)
        }
        
    except Exception as e:
        return {
            'statusCode': 500,
            'body': json.dumps({'error': str(e)})
        }


def scrape_epic_games():
    """Scraping com Playwright otimizado para Lambda"""
    
    with sync_playwright() as p:
        browser = p.chromium.launch(
            headless=True,
            args=[
                '--no-sandbox',
                '--disable-dev-shm-usage',
                '--disable-gpu',
                '--single-process'  # Economiza memória
            ]
        )
        
        context = browser.new_context(
            viewport={'width': 1280, 'height': 720},
            user_agent='Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
        )
        
        page = context.new_page()
        
        try:
            page.goto(
                'https://store.epicgames.com/pt-BR/free-games',
                wait_until='networkidle',
                timeout=30000
            )
            
            page.wait_for_selector('section', timeout=10000)
            
            games = page.evaluate('''() => {
                const gameLinks = Array.from(document.querySelectorAll('a[href*="/p/"]'));
                return gameLinks.map(link => {
                    const title = link.querySelector('h3, h2');
                    const img = link.querySelector('img');
                    
                    return {
                        title: title ? title.textContent.trim() : 'Sem título',
                        url: link.href,
                        image: img ? img.src : null
                    };
                }).filter(game => game.title !== 'Sem título');
            }''')
            
            return games
            
        finally:
            browser.close()


if __name__ == "__main__":
    result = lambda_handler({}, {})
    print(json.dumps(json.loads(result['body']), indent=2))

"""
Epic Games Free Games Scraper - AWS Lambda Function
Versão Production-Ready com API GraphQL Oficial

Configuração Lambda Recomendada:
- Runtime: Python 3.12
- Memória: 128 MB
- Timeout: 10 segundos
- Nenhuma Layer necessária (usa apenas stdlib)

Custo Estimado: FREE TIER permanente
- ~1 segundo por execução
- Sem dependências externas
- 128MB suficiente

Autor: Jonathas Rocha
Data: 2025-11-17
"""

import json
import urllib.request
import urllib.error
from datetime import datetime
from typing import Dict, List, Any, Optional
import logging

logger = logging.getLogger()
logger.setLevel(logging.INFO)


def lambda_handler(event: Dict[str, Any], context: Any) -> Dict[str, Any]:
    """
    Handler principal da AWS Lambda Function
    
    Args:
        event: Evento Lambda (não utilizado nesta versão)
        context: Contexto Lambda com informações de runtime
        
    Returns:
        Dict com statusCode, headers e body (JSON)
        
    Exemplo de Resposta Sucesso:
        {
            "statusCode": 200,
            "headers": {...},
            "body": "{\"scraped_at\": \"2025-11-17T...\", \"total_games\": 2, ...}"
        }
        
    Exemplo de Resposta Erro:
        {
            "statusCode": 500,
            "headers": {...},
            "body": "{\"error\": \"Mensagem de erro\", \"details\": \"...\"}"
        }
    """
    logger.info("🚀 Iniciando Epic Games Free Scraper")
    logger.info(f"Request ID: {context.request_id if context else 'local'}")
    
    try:
        games = get_free_games_api()
        
        response_body = {
            'scraped_at': datetime.now().isoformat(),
            'total_games': len(games),
            'games': games,
            'source': 'Epic Games Store API',
            'locale': 'pt-BR'
        }
        
        logger.info(f"✅ Scraping concluído: {len(games)} jogos encontrados")
        
        return {
            'statusCode': 200,
            'headers': {
                'Content-Type': 'application/json; charset=utf-8',
                'Access-Control-Allow-Origin': '*',
                'Access-Control-Allow-Methods': 'GET, OPTIONS',
                'Access-Control-Allow-Headers': 'Content-Type',
                'Cache-Control': 'max-age=3600'  # Cache de 1 hora
            },
            'body': json.dumps(response_body, ensure_ascii=False, indent=2)
        }
        
    except urllib.error.HTTPError as e:
        logger.error(f"❌ HTTP Error: {e.code} - {e.reason}")
        return {
            'statusCode': 502,
            'headers': {'Content-Type': 'application/json'},
            'body': json.dumps({
                'error': 'Erro ao acessar API da Epic Games',
                'details': f'HTTP {e.code}: {e.reason}',
                'timestamp': datetime.now().isoformat()
            })
        }
        
    except urllib.error.URLError as e:
        logger.error(f"❌ URL Error: {e.reason}")
        return {
            'statusCode': 503,
            'headers': {'Content-Type': 'application/json'},
            'body': json.dumps({
                'error': 'Erro de conexão com Epic Games',
                'details': str(e.reason),
                'timestamp': datetime.now().isoformat()
            })
        }
        
    except json.JSONDecodeError as e:
        logger.error(f"❌ JSON Decode Error: {e}")
        return {
            'statusCode': 502,
            'headers': {'Content-Type': 'application/json'},
            'body': json.dumps({
                'error': 'Erro ao processar resposta da API',
                'details': f'JSON inválido na linha {e.lineno}',
                'timestamp': datetime.now().isoformat()
            })
        }
        
    except Exception as e:
        logger.error(f"❌ Erro inesperado: {type(e).__name__}: {e}")
        return {
            'statusCode': 500,
            'headers': {'Content-Type': 'application/json'},
            'body': json.dumps({
                'error': 'Erro interno do servidor',
                'details': str(e),
                'type': type(e).__name__,
                'timestamp': datetime.now().isoformat()
            })
        }


def get_free_games_api() -> List[Dict[str, Any]]:
    """
    Busca jogos gratuitos usando a API GraphQL oficial da Epic Games
    
    Returns:
        Lista de dicionários com informações dos jogos gratuitos
        
    Raises:
        urllib.error.HTTPError: Se a API retornar erro HTTP
        urllib.error.URLError: Se houver erro de conexão
        json.JSONDecodeError: Se a resposta não for JSON válido
        
    Estrutura do Retorno:
        [
            {
                'title': str,
                'description': str,
                'url': str,
                'slug': str,
                'original_price': str,
                'discount_price': str,
                'publisher': str,
                'developer': str,
                'thumbnail': str (opcional),
                'banner': str (opcional),
                'start_date': str (ISO 8601),
                'end_date': str (ISO 8601),
                'discount_percentage': int
            },
            ...
        ]
    """
    api_url = (
        "https://store-site-backend-static.ak.epicgames.com/"
        "freeGamesPromotions?locale=pt-BR&country=BR&allowCountries=BR"
    )
    
    logger.info(f"📡 Acessando API: {api_url}")
    
    request = urllib.request.Request(
        api_url,
        headers={
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/119.0.0.0 Safari/537.36',
            'Accept': 'application/json',
            'Accept-Language': 'pt-BR,pt;q=0.9,en-US;q=0.8,en;q=0.7',
            'Accept-Encoding': 'gzip, deflate, br',
            'Connection': 'keep-alive'
        }
    )
    
    with urllib.request.urlopen(request, timeout=10) as response:
        data = json.loads(response.read().decode('utf-8'))
    
    logger.info("✅ API respondeu com sucesso")
    
    elements = data.get('data', {}).get('Catalog', {}).get('searchStore', {}).get('elements', [])
    logger.info(f"📦 Total de elementos no catálogo: {len(elements)}")
    
    games = []
    
    for game in elements:
        promotions = game.get('promotions', {})
        promotional_offers = promotions.get('promotionalOffers', [])
        
        if not promotional_offers:
            continue
        
        game_data = {
            'title': game.get('title', 'Sem título'),
            'description': game.get('description', '').strip(),
            'slug': game.get('productSlug', ''),
            'url': f"https://store.epicgames.com/pt-BR/p/{game.get('productSlug', '')}",
            'original_price': _extract_price(game, 'originalPrice'),
            'discount_price': _extract_price(game, 'discountPrice'),
            'publisher': game.get('seller', {}).get('name', 'Epic Games'),
            'developer': _extract_custom_attribute(game, 'developerName'),
            'discount_percentage': _calculate_discount_percentage(game)
        }
        
        key_images = game.get('keyImages', [])
        for img in key_images:
            img_type = img.get('type')
            img_url = img.get('url')
            
            if img_type == 'Thumbnail' and img_url:
                game_data['thumbnail'] = img_url
            elif img_type == 'DieselStoreFrontWide' and img_url:
                game_data['banner'] = img_url
            elif img_type == 'OfferImageWide' and img_url and 'banner' not in game_data:
                game_data['banner'] = img_url
        
        if promotional_offers and promotional_offers[0].get('promotionalOffers'):
            offer = promotional_offers[0]['promotionalOffers'][0]
            game_data['start_date'] = offer.get('startDate', '')
            game_data['end_date'] = offer.get('endDate', '')
            
            if game_data['end_date']:
                game_data['end_date_formatted'] = _format_datetime(game_data['end_date'])
        
        games.append(game_data)
        logger.info(f"  ✓ {game_data['title']}")
    
    logger.info(f"🎮 Total de jogos gratuitos ativos: {len(games)}")
    
    return games


def _extract_price(game: Dict, price_type: str) -> str:
    """
    Extrai preço formatado do jogo
    
    Args:
        game: Dicionário com dados do jogo
        price_type: 'originalPrice' ou 'discountPrice'
        
    Returns:
        Preço formatado como string (ex: "R$ 99,99" ou "Grátis")
    """
    try:
        price = game.get('price', {}).get('totalPrice', {}).get('fmtPrice', {}).get(price_type)
        if price == '0':
            return 'Grátis'
        return price or 'N/A'
    except (KeyError, AttributeError):
        return 'N/A'


def _extract_custom_attribute(game: Dict, attribute_name: str) -> Optional[str]:
    """
    Extrai atributo customizado do jogo
    
    Args:
        game: Dicionário com dados do jogo
        attribute_name: Nome do atributo a extrair
        
    Returns:
        Valor do atributo ou None
    """
    try:
        custom_attributes = game.get('customAttributes', [])
        for attr in custom_attributes:
            if attr.get('key') == attribute_name:
                return attr.get('value')
        return None
    except (KeyError, AttributeError):
        return None


def _calculate_discount_percentage(game: Dict) -> int:
    """
    Calcula percentual de desconto
    
    Args:
        game: Dicionário com dados do jogo
        
    Returns:
        Percentual de desconto como inteiro (0-100)
    """
    try:
        price_info = game.get('price', {}).get('totalPrice', {})
        original = price_info.get('originalPrice', 0)
        discount = price_info.get('discountPrice', 0)
        
        if original > 0:
            percentage = int(((original - discount) / original) * 100)
            return percentage
        return 0
    except (KeyError, AttributeError, ZeroDivisionError, TypeError):
        return 0


def _format_datetime(iso_datetime: str) -> str:
    """
    Formata datetime ISO 8601 para formato brasileiro
    
    Args:
        iso_datetime: String ISO 8601 (ex: "2025-11-21T15:00:00.000Z")
        
    Returns:
        Data formatada (ex: "21/11/2025 às 12:00")
    """
    try:
        dt = datetime.fromisoformat(iso_datetime.replace('Z', '+00:00'))
        from datetime import timedelta
        dt_br = dt - timedelta(hours=3)
        return dt_br.strftime("%d/%m/%Y às %H:%M")
    except (ValueError, AttributeError):
        return iso_datetime

📤 Deploy na AWS Lambda

# Opção 1: Via Console AWS (Mais Fácil)
1. Acesse AWS Lambda Console
2. Clique em "Create function"
3. Nome: epic-games-scraper
4. Runtime: Python 3.12
5. Clique em "Create function"
6. Cole o código na aba "Code"
7. Clique em "Deploy"
8. Configure:
   - Memory: 128 MB
   - Timeout: 10 seconds

zip lambda.zip lambda_function.py

aws lambda create-function \
    --function-name epic-games-scraper \
    --runtime python3.12 \
    --role arn:aws:iam::YOUR_ACCOUNT:role/lambda-basic-execution \
    --handler lambda_function.lambda_handler \
    --zip-file fileb://lambda.zip \
    --timeout 10 \
    --memory-size 128

service: epic-games-scraper

provider:
  name: aws
  runtime: python3.12
  region: us-east-1
  memorySize: 128
  timeout: 10

functions:
  scraper:
    handler: lambda_function.lambda_handler
    events:
      - http:
          path: free-games
          method: get
          cors: true

🎉 Campos Retornados por Jogo:
  • title: Título do jogo
  • description: Descrição completa
  • slug: Identificador único na URL
  • url: Link direto para a página do jogo
  • original_price: Preço original formatado (ex: "R$ 199,99")
  • discount_price: Preço com desconto (sempre "Grátis")
  • publisher: Nome da editora/publicadora
  • developer: Nome do desenvolvedor
  • thumbnail: URL da imagem miniatura (quadrada)
  • banner: URL da imagem banner (widescreen)
  • start_date: Data de início da promoção (ISO 8601)
  • end_date: Data de fim da promoção (ISO 8601)
  • end_date_formatted: Data formatada em pt-BR
  • discount_percentage: Percentual de desconto (0-100)

🔧 Versão Avançada (com Selenium)

Para sites com muito JavaScript, use esta versão com Selenium: 

#!/usr/bin/env python3
"""
Epic Games Free Games Scraper - Versão Selenium
Mais robusto para páginas com JavaScript pesado
"""

from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.chrome.options import Options
import json
from datetime import datetime
import time


def setup_driver():
    """Configura o driver do Selenium"""
    options = Options()
    options.add_argument('--headless')  # Roda sem interface gráfica
    options.add_argument('--no-sandbox')
    options.add_argument('--disable-dev-shm-usage')
    options.add_argument('--disable-blink-features=AutomationControlled')
    options.add_argument('user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36')
    
    driver = webdriver.Chrome(options=options)
    return driver


def scrape_epic_games_selenium():
    """Scraping com Selenium para melhor compatibilidade"""
    url = "https://store.epicgames.com/pt-BR/free-games"
    driver = setup_driver()
    games = []
    
    try:
        print(f"🔍 Acessando: {url}")
        driver.get(url)
        
        # Aguarda a página carregar
        print("⏳ Aguardando carregamento...")
        time.sleep(5)  # Aguarda JavaScript executar
        
        # Aguarda elementos aparecerem
        wait = WebDriverWait(driver, 15)
        wait.until(EC.presence_of_element_located((By.TAG_NAME, "section")))
        
        # Busca a section
        sections = driver.find_elements(By.TAG_NAME, "section")
        print(f"📦 Encontradas {len(sections)} sections")
        
        # Busca cards de jogos
        game_elements = driver.find_elements(By.CSS_SELECTOR, "a[href*='/p/']")
        print(f"🎮 Encontrados {len(game_elements)} jogos")
        
        for idx, element in enumerate(game_elements, 1):
            try:
                game_data = {}
                
                # Título
                try:
                    title = element.find_element(By.TAG_NAME, "h3")
                    game_data['title'] = title.text
                except:
                    try:
                        title = element.find_element(By.TAG_NAME, "h2")
                        game_data['title'] = title.text
                    except:
                        game_data['title'] = "Título não encontrado"
                
                # URL
                game_data['url'] = element.get_attribute('href')
                
                # Imagem
                try:
                    img = element.find_element(By.TAG_NAME, "img")
                    game_data['image'] = img.get_attribute('src')
                except:
                    pass
                
                if game_data.get('title') and game_data['title'] != "Título não encontrado":
                    games.append(game_data)
                    print(f"  {idx}. {game_data['title']}")
                    
            except Exception as e:
                print(f"⚠️  Erro ao processar jogo {idx}: {e}")
                continue
        
        # Salvar em JSON
        output = {
            'scraped_at': datetime.now().isoformat(),
            'total_games': len(games),
            'games': games
        }
        
        with open('epic_free_games.json', 'w', encoding='utf-8') as f:
            json.dump(output, f, ensure_ascii=False, indent=2)
        
        print(f"\n✅ Dados salvos em: epic_free_games.json")
        print(f"📊 Total de jogos: {len(games)}")
        
    finally:
        driver.quit()


if __name__ == "__main__":
    scrape_epic_games_selenium()
